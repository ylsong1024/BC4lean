  5 /-!
  6 # The complex group algebra of an arbitrary group
  7 
  8 This file verifies the algebraic starting point for the reduced group C*-algebra.
  9 Multiplication is Mathlib's convolution product, not pointwise multiplication.
 10 The involution conjugates coefficients and reverses group elements.
 11 No topology, countability, regular representation, or completion is assumed here.
 12 -/

 25 /-- The basis vector at a group element. -/
 26 def delta (g : Γ) : GroupAlgebra Γ := MonoidAlgebra.single g 1

 28 /-- The finite coefficient ℓ¹-norm; no normed-algebra instance is claimed. -/
 29 def l1Norm (a : GroupAlgebra Γ) : ℝ := ∑ g ∈ a.coeff.support, ‖a.coeff g‖

 31 /-- Conjugation of coefficients followed by inversion of the group index. -/
 32 def involution (a : GroupAlgebra Γ) : GroupAlgebra Γ :=
 33   .ofCoeff (Finsupp.equivMapDomain (Equiv.inv Γ)
 34     (a.coeff.mapRange star (star_zero ℂ)))

 61 @[simp]
 62 theorem involution_single (g : Γ) (z : ℂ) :
 63     involution (MonoidAlgebra.single g z) = MonoidAlgebra.single g⁻¹ (star z) := by
 64   simp only [involution, MonoidAlgebra.coeff_single, Finsupp.mapRange_single,

 67 theorem involution_mul (a b : GroupAlgebra Γ) :
 68     involution (a * b) = involution b * involution a := by
 69   induction a using MonoidAlgebra.induction_linear with
 70   | zero => simp
 71   | add a c ha hc => simp only [add_mul, involution_add, ha, hc, mul_add]
 72   | single g z =>
 73     induction b using MonoidAlgebra.induction_linear with
 74     | zero => simp
 75     | add b c hb hc => simp only [mul_add, involution_add, hb, hc, add_mul]
 76     | single h w => simp [MonoidAlgebra.single_mul_single, mul_inv_rev, mul_comm]

 78 /-- The group-algebra star is conjugation combined with inversion. -/
 79 instance instStar : Star (GroupAlgebra Γ) := ⟨involution⟩
 80 
 81 /-- Convolution and the group-algebra involution form a star ring. -/
 82 instance instStarRing : StarRing (GroupAlgebra Γ) where
 83   star_involutive := involution_involution
 84   star_mul := involution_mul
 85   star_add := involution_add
 86 
 87 /-- The involution is conjugate-linear over the complex scalars. -/
 88 instance instStarModule : StarModule ℂ (GroupAlgebra Γ) where
 89   star_smul := involution_smul

 91 @[simp]
 92 theorem coeff_star (a : GroupAlgebra Γ) (g : Γ) :
 93     (star a).coeff g = star (a.coeff g⁻¹) := coeff_involution a g

 95 /-- The multiplication agrees with the blueprint's convolution formula. -/
 96 theorem coeff_convolution (a b : GroupAlgebra Γ) (h : Γ) :
 97     (a * b).coeff h = ∑ g ∈ a.coeff.support, a.coeff g * b.coeff (g⁻¹ * h) :=
 98   MonoidAlgebra.coeff_mul_apply_left a b h

121 theorem support_star (a : GroupAlgebra Γ) :
122     (star a).coeff.support = a.coeff.support.map (Equiv.inv Γ).toEmbedding := by
123   classical
124   ext g

133 theorem convolution_bilinear (a b c : GroupAlgebra Γ) (z : ℂ) :
134     (a + b) * c = a * c + b * c ∧ a * (b + c) = a * b + a * c ∧
135     (z • a) * b = z • (a * b) ∧ a * (z • b) = z • (a * b) :=
136   ⟨add_mul a b c, mul_add a b c, smul_mul_assoc z a b, mul_smul_comm z a b⟩

138 /-- The laws appearing together in the blueprint's algebraic proposition. -/
139 theorem star_algebra_laws (a b c : GroupAlgebra Γ) (z w : ℂ) :
140     (a * b) * c = a * (b * c) ∧
141     (delta 1) * a = a ∧ a * (delta 1) = a ∧
142     star (a * b) = star b * star a ∧ star (star a) = a ∧
143     star (z • a + w • b) = star z • star a + star w • star b := by
144   simp only [mul_assoc, delta_one, one_mul, mul_one, star_mul, star_star,
145     star_add, star_smul, and_self]
