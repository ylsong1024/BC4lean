\chapter{Proper actions and the universal proper space}
\label{ch:proper-actions}

This chapter begins with the verified topology of proper discrete-group
actions and their subgroup fixed-point spaces. Continuous equivariant maps,
homotopies, and their fixed-point restrictions are also verified. The results below
add homogeneous orbits, proper orbit cells, the fixed-point criterion, a finite-group
point model, conditional uniqueness, and directed cocompact pieces for a given space.
Equivariant CW attachment, the general existence and universal property of
$\underline E\Gamma$, and the locally compact second-countable model choice
remain formalization targets.

\section{Proper actions and compact transporters}

\begin{definition}\label{def:proper-action}
\lean{BC4lean.ProperActions.actionMap,BC4lean.ProperActions.proper_iff_actionMap}
\leanok
For a topological group $\Gamma$ acting on a topological space $X$, the
action is proper when
\[
 \alpha:\Gamma\times X\longrightarrow X\times X,
 \qquad \alpha(g,x)=(gx,x)
\]
is a proper map. This is exactly Mathlib's \texttt{ProperSMul} predicate;
it includes continuity of the action. In the applications below $\Gamma$
is discrete. Finite stabilizers alone do not replace properness.
\end{definition}

\begin{definition}\label{def:compact-transporter}
\lean{BC4lean.ProperActions.transporter}
\leanok
For subsets $K,L\subseteq X$, define the transporter
\[
 T(K,L)=\{g\in\Gamma:gK\cap L\ne\varnothing\}.
\]
\end{definition}

\begin{proposition}\label{prop:compact-transporter-criterion}
\uses{def:proper-action,def:compact-transporter}
\lean{BC4lean.ProperActions.finite_transporter,BC4lean.ProperActions.proper_iff_finite_transporters,BC4lean.ProperActions.proper_iff_compact_preimages}
For a proper action of a discrete group, $T(K,L)$ is finite whenever
$K,L$ are compact. If $X$ is locally compact Hausdorff and the action is
continuous, the converse holds. Under these hypotheses properness is also
equivalent to compactness of $\alpha^{-1}(C)$ for every compact
$C\subseteq X\times X$.
\end{proposition}
\begin{proof}
\leanok
The first projection maps the compact set $\alpha^{-1}(L\times K)$
onto $T(K,L)$. Compact subsets of a discrete space are finite.
Conversely, for compact $C\subseteq X\times X$, its coordinate projections
are compact. The inverse image of $C$ is a closed subset of the product
of a finite transporter and the second coordinate projection.
The pinned Mathlib proper-map and compact-transporter criteria formalize
these implications on locally compact Hausdorff spaces.
\end{proof}

\begin{proposition}\label{prop:proper-finite-stabilizers}
\uses{prop:compact-transporter-criterion}
\lean{BC4lean.ProperActions.finite_stabilizer}
Every point stabilizer $\Gamma_x$ in a proper discrete-group action is
finite. This implication does not require local compactness of $X$.
\end{proposition}
\begin{proof}
\leanok
The stabilizer is $T(\{x\},\{x\})$, and singletons are compact.
\end{proof}

\begin{proposition}\label{prop:proper-orbit-space}
\uses{def:proper-action}
\lean{BC4lean.ProperActions.orbitSpace_t2,BC4lean.ProperActions.orbitSpace_locallyCompact}
A proper action has Hausdorff orbit space $X/\Gamma$. If $X$ is locally
compact, then $X/\Gamma$ is locally compact as well.
\end{proposition}
\begin{proof}
\leanok
Use Mathlib's Hausdorff quotient theorem for proper actions. The orbit
projection is an open quotient map for a continuous group action, and
local compactness descends through an open quotient map.
\end{proof}

\begin{proposition}\label{prop:proper-subgroup-restriction}
\uses{def:proper-action}
\lean{BC4lean.ProperActions.proper_subgroup}
Restricting a proper discrete-group action to any subgroup preserves
properness.
\end{proposition}
\begin{proof}
\leanok
Every subgroup of a discrete group is closed. Apply the proper-action
restriction theorem for a closed embedding of groups.
\end{proof}

\begin{proposition}\label{prop:proper-action-examples}
\uses{prop:compact-transporter-criterion}
\lean{BC4lean.ProperActions.proper_of_finite,BC4lean.ProperActions.proper_leftTranslation,BC4lean.ProperActions.finite_group_of_compact}
Every continuous action of a finite discrete group on a locally compact
Hausdorff space is proper. Left translation of a topological group on
itself is proper. If a discrete group acts properly on a nonempty compact
space, the group is finite.
\end{proposition}
\begin{proof}
\leanok
For a finite group every transporter is finite. For left translation,
$(g,x)\mapsto(gx,x)$ is a homeomorphism with inverse
$(y,x)\mapsto(yx^{-1},x)$. For the last assertion,
$T(X,X)=\Gamma$ is finite by compactness and nonemptiness.
\end{proof}

\section{Subgroup fixed-point spaces}

\begin{definition}\label{def:subgroup-fixed-point-space}
\lean{BC4lean.ProperActions.FixedPointSpace}
\leanok
For a subgroup $H\leq\Gamma$, let
\[
 X^H=\{x\in X: hx=x\text{ for every }h\in H\}
\]
with the subspace topology. The Lean definition uses Mathlib's existing
\texttt{MulAction.fixedPoints}.
\end{definition}

\begin{proposition}\label{prop:fixed-point-subgroup-laws}
\uses{def:subgroup-fixed-point-space}
\lean{BC4lean.ProperActions.mem_fixedPoints_iff_le_stabilizer,BC4lean.ProperActions.fixedPoints_antitone,BC4lean.ProperActions.fixedPoints_bot}
A point $x$ lies in $X^H$ exactly when $H\leq\Gamma_x$.
If $H\leq K$, then $X^K\subseteq X^H$, and $X^{\{1\}}=X$.
\end{proposition}
\begin{proof}
\leanok
Unfold the fixed-point and stabilizer conditions and use subgroup inclusion.
\end{proof}

\begin{proposition}\label{prop:fixed-point-space-closed}
\uses{def:subgroup-fixed-point-space}
\lean{BC4lean.ProperActions.isClosed_fixedPoints,BC4lean.ProperActions.fixedPointSpace_locallyCompact}
For an action by continuous translations on a Hausdorff space, $X^H$
is closed for every subgroup $H$. If $X$ is locally compact, so is $X^H$.
\end{proposition}
\begin{proof}
\leanok
The fixed-point set is the intersection, over $h\in H$, of the closed
equalizers of $x\mapsto hx$ and $x\mapsto x$. Closed subspaces of locally
compact spaces are locally compact.
\end{proof}

\begin{proposition}\label{prop:proper-infinite-fixed-points}
\uses{prop:proper-finite-stabilizers,prop:fixed-point-subgroup-laws}
\lean{BC4lean.ProperActions.finite_subgroup_of_fixedPoint,BC4lean.ProperActions.fixedPoints_eq_empty_of_infinite,BC4lean.ProperActions.fixedPointSpace_isEmpty}
For a proper discrete-group action, any subgroup fixing a point is finite.
Consequently $X^H=\varnothing$ whenever $H$ is infinite.
\end{proposition}
\begin{proof}
\leanok
If $x\in X^H$, then $H$ is a subgroup of the finite group $\Gamma_x$.
\end{proof}

Properness does not imply contractibility of fixed-point spaces for finite
subgroups. That is additional structure required of a universal proper model.

\section{Equivariant maps and homotopies}

\begin{definition}\label{def:equivariant-map}
\lean{BC4lean.ProperActions.EquivariantMap,BC4lean.ProperActions.EquivariantMap.toMulActionHom}
\leanok
For topological spaces $X,Y$ with $\Gamma$-actions, a continuous equivariant
map $f:X\to Y$ is a continuous map satisfying $f(gx)=gf(x)$.
The underlying equivariant function uses Mathlib's \texttt{MulActionHom}.
The constructions in this section do not require properness, a discrete
group topology, local compactness, or Hausdorffness.
\end{definition}

\begin{proposition}\label{prop:equivariant-map-laws}
\uses{def:equivariant-map}
\lean{BC4lean.ProperActions.EquivariantMap.ext,BC4lean.ProperActions.EquivariantMap.map_smul,BC4lean.ProperActions.EquivariantMap.continuous,BC4lean.ProperActions.EquivariantMap.id,BC4lean.ProperActions.EquivariantMap.comp,BC4lean.ProperActions.EquivariantMap.id_apply,BC4lean.ProperActions.EquivariantMap.comp_apply,BC4lean.ProperActions.EquivariantMap.id_comp,BC4lean.ProperActions.EquivariantMap.comp_id,BC4lean.ProperActions.EquivariantMap.comp_assoc}
Continuous equivariant maps have identity and associative composition, with
both identity laws. Equality of maps is determined pointwise.
\end{proposition}
\begin{proof}
\leanok
Use the identity and composition of continuous maps. Equivariance is
preserved because $g(f(ax))=g(af(x))=ag(f(x))$. The laws follow pointwise.
\end{proof}

\begin{proposition}\label{prop:fixed-point-map-functoriality}
\uses{prop:equivariant-map-laws,def:subgroup-fixed-point-space}
\lean{BC4lean.ProperActions.EquivariantMap.map_fixedPoint,BC4lean.ProperActions.EquivariantMap.fixedPointsMap,BC4lean.ProperActions.EquivariantMap.fixedPointsMap_apply,BC4lean.ProperActions.EquivariantMap.fixedPointsMap_id,BC4lean.ProperActions.EquivariantMap.fixedPointsMap_comp}
For every subgroup $H\leq\Gamma$, an equivariant map $f:X\to Y$ restricts
to a continuous map $f^H:X^H\to Y^H$. Restriction preserves identity and
composition. The subgroup need not be normal.
\end{proposition}
\begin{proof}
\leanok
For $h\in H$ and $x\in X^H$, one has $hf(x)=f(hx)=f(x)$.
Continuity follows from the subspace topology; the laws follow pointwise.
\end{proof}

\begin{definition}\label{def:equivariant-homotopy}
\uses{def:equivariant-map}
\lean{BC4lean.ProperActions.EquivariantHomotopy,BC4lean.ProperActions.EquivariantlyHomotopic}
\leanok
An equivariant homotopy from $f_0$ to $f_1$ is a jointly continuous map
$F:[0,1]\times X\to Y$ with $F(0,x)=f_0(x)$, $F(1,x)=f_1(x)$, and
$F(t,gx)=gF(t,x)$. The interval has trivial action.
Write $f_0\simeq_\Gamma f_1$ when such a homotopy exists.
The Lean definition specializes Mathlib's \texttt{ContinuousMap.HomotopyWith}.
\end{definition}

\begin{proposition}\label{prop:equivariant-homotopy-operations}
\uses{def:equivariant-homotopy,prop:equivariant-map-laws}
\lean{BC4lean.ProperActions.EquivariantHomotopy.refl,BC4lean.ProperActions.EquivariantHomotopy.symm,BC4lean.ProperActions.EquivariantHomotopy.trans,BC4lean.ProperActions.EquivariantHomotopy.comp,BC4lean.ProperActions.EquivariantHomotopy.atTime,BC4lean.ProperActions.EquivariantHomotopy.at_apply,BC4lean.ProperActions.EquivariantHomotopy.at_zero,BC4lean.ProperActions.EquivariantHomotopy.at_one,BC4lean.ProperActions.EquivariantlyHomotopic.refl,BC4lean.ProperActions.EquivariantlyHomotopic.symm,BC4lean.ProperActions.EquivariantlyHomotopic.trans,BC4lean.ProperActions.EquivariantlyHomotopic.comp}
Equivariant homotopy is reflexive, symmetric, transitive, and compatible
with composition. Every time slice of an equivariant homotopy is a
continuous equivariant map, with the specified maps at times zero and one.
\end{proposition}
\begin{proof}
\leanok
Use constant homotopies, time reversal, and concatenation on the two half
intervals. Composition is $(G\circ F)(t,x)=G(t,F(t,x))$.
Joint continuity follows from Mathlib's homotopy operations; equivariance
holds at each time. Restriction to a time slice is continuous.
\end{proof}

\begin{proposition}\label{prop:fixed-point-homotopy-restriction}
\uses{prop:equivariant-homotopy-operations,prop:fixed-point-map-functoriality}
\lean{BC4lean.ProperActions.EquivariantHomotopy.onFixedPoints,BC4lean.ProperActions.EquivariantHomotopy.onFixedPoints_apply,BC4lean.ProperActions.EquivariantHomotopy.onFixedPoints_refl,BC4lean.ProperActions.EquivariantHomotopy.onFixedPoints_symm,BC4lean.ProperActions.EquivariantHomotopy.onFixedPoints_trans,BC4lean.ProperActions.EquivariantHomotopy.onFixedPoints_comp,BC4lean.ProperActions.EquivariantlyHomotopic.onFixedPoints}
An equivariant homotopy $F:f_0\simeq_\Gamma f_1$ restricts to an ordinary
continuous homotopy $F^H:f_0^H\simeq f_1^H$ for every subgroup $H$.
Restriction preserves constant homotopies, reversal, concatenation with
the chosen half-interval parametrization, and composition of homotopies.
\end{proposition}
\begin{proof}
\leanok
At each time equivariance ensures $F(t,X^H)\subseteq Y^H$.
The map into $Y^H$ is jointly continuous by the subspace topology, and the
endpoint identities hold as equalities of subtype elements. The operation
laws follow from their formulas, splitting the two time intervals for
concatenation.
\end{proof}

\begin{definition}\label{def:equivariant-homotopy-equivalence}
\uses{def:equivariant-homotopy,prop:equivariant-map-laws}
\lean{BC4lean.ProperActions.EquivariantHomotopyEquiv}
\leanok
An equivariant homotopy equivalence between $X$ and $Y$ consists of
continuous equivariant maps $f:X\to Y$ and $g:Y\to X$ with
$gf\simeq_\Gamma\operatorname{id}_X$ and
$fg\simeq_\Gamma\operatorname{id}_Y$.
\end{definition}

\begin{proposition}\label{prop:fixed-point-homotopy-equivalence}
\uses{def:equivariant-homotopy-equivalence,prop:fixed-point-homotopy-restriction}
\lean{BC4lean.ProperActions.EquivariantHomotopyEquiv.refl,BC4lean.ProperActions.EquivariantHomotopyEquiv.symm,BC4lean.ProperActions.EquivariantHomotopyEquiv.trans,BC4lean.ProperActions.EquivariantHomotopyEquiv.fixedPoints,BC4lean.ProperActions.EquivariantHomotopyEquiv.fixedPoints_toFun_apply,BC4lean.ProperActions.EquivariantHomotopyEquiv.fixedPoints_invFun_apply}
Equivariant homotopy equivalences have identity, reversal, and composition.
Every equivariant homotopy equivalence $X\simeq_\Gamma Y$ induces an
ordinary homotopy equivalence $X^H\simeq Y^H$ for every subgroup $H$.
\end{proposition}
\begin{proof}
\leanok
Compose the equivariant maps and their inverse homotopies to obtain the
three operations. Restrict both maps and both inverse homotopies to the
fixed-point spaces, using the proved identity and composition laws.
\end{proof}

\section{Homogeneous orbits and orbit cells}

\begin{proposition}\label{prop:orbit-fixed-point-correspondence}
\uses{def:equivariant-map,def:subgroup-fixed-point-space}
\lean{BC4lean.ProperActions.orbitFixedPointEquiv,BC4lean.ProperActions.orbitFixedPointEquiv_natural}
For a discrete group $\Gamma$, any subgroup $H$, and any $\Gamma$-space $X$,
evaluation at $H\in\Gamma/H$ is a bijection
\[
 \operatorname{Map}_\Gamma(\Gamma/H,X)\cong X^H
\]
for continuous equivariant maps. Its inverse sends $x$ to $gH\mapsto gx$.
The bijection is natural under continuous equivariant maps in $X$.
No normality or finiteness hypothesis on $H$ is required.
\end{proposition}
\begin{proof}
\leanok
The fixed-point condition makes the orbit map independent of the coset
representative. The orbit is discrete, so the descended map is continuous.
Equivariance determines a map by its value at the identity coset.
\end{proof}

\begin{proposition}\label{prop:proper-homogeneous-orbits}
\uses{prop:proper-finite-stabilizers,prop:orbit-fixed-point-correspondence}
\lean{BC4lean.ProperActions.proper_cosetProjection,BC4lean.ProperActions.proper_orbit_iff}
The left action on $\Gamma/H$ is proper if and only if $H$ is finite.
When $H$ is finite, the projection $\Gamma\to\Gamma/H$ is a proper map.
\end{proposition}
\begin{proof}
\leanok
Fibers of the coset projection are finite translates of $H$. A continuous
map between discrete spaces is closed, so this projection is proper.
The proper-orbit-map criterion gives properness of the action.
Conversely, $H$ fixes the identity coset and hence must be finite.
\end{proof}

\begin{definition}\label{def:orbit-cell}
\uses{def:equivariant-map}
\lean{BC4lean.ProperActions.OrbitCell,BC4lean.ProperActions.OrbitCell.projection}
\leanok
For a subgroup $H$ and a space $D$, the orbit cell is
$\Gamma/H\times D$ with action $g(q,d)=(gq,d)$ and the product topology.
Projection onto $\Gamma/H$ is continuous and equivariant.
Taking $D$ to be a disk gives the cell shape for the intended CW construction;
cell attachment is a separate target below.
\end{definition}

\begin{proposition}\label{prop:proper-orbit-cells}
\uses{def:orbit-cell,prop:proper-homogeneous-orbits,prop:compact-transporter-criterion}
\lean{BC4lean.ProperActions.proper_of_equivariantMap,BC4lean.ProperActions.OrbitCell.proper}
A continuous equivariant map from a locally compact Hausdorff
$\Gamma$-space to a proper $\Gamma$-space implies properness of the source.
In particular, $\Gamma/H\times D$ is proper if $H$ is finite and $D$ is
locally compact Hausdorff.
\end{proposition}
\begin{proof}
\leanok
The transporter of two compact subsets of the source is contained in
the transporter of their compact images. Apply the finite-transporter
criterion, and then apply the result to the orbit-cell projection.
\end{proof}

\section{The fixed-point criterion and uniqueness}

\begin{definition}\label{def:proper-model-fixed-point-criterion}
\uses{def:subgroup-fixed-point-space}
\lean{BC4lean.ProperActions.FixedPointCriterion}
\leanok
The fixed-point criterion for a prospective model $E$ requires $E^H$ to
be contractible for every finite subgroup $H$ and empty for every infinite
subgroup $H$. This predicate alone does not supply an equivariant CW
structure or a proof of the universal mapping property.
\end{definition}

\begin{proposition}\label{prop:fixed-point-criterion-invariance}
\uses{def:proper-model-fixed-point-criterion,prop:fixed-point-homotopy-equivalence}
\lean{BC4lean.ProperActions.FixedPointCriterion.of_equiv}
The fixed-point criterion is preserved by equivariant homotopy equivalence.
\end{proposition}
\begin{proof}
\leanok
Restrict an equivariant homotopy equivalence to each subgroup's fixed
points. Contractibility is invariant under homotopy equivalence, and the
inverse map transfers emptiness.
\end{proof}

\begin{definition}\label{def:homotopy-terminal-for}
\uses{def:equivariant-map,def:equivariant-homotopy}
\lean{BC4lean.ProperActions.HomotopyTerminalFor}
\leanok
For a specified source $X$ and target $Y$, the homotopy-terminal mapping
condition says that a continuous equivariant map $X\to Y$ exists and any
two such maps are equivariantly homotopic.
\end{definition}

\begin{proposition}\label{prop:conditional-universal-model-uniqueness}
\uses{def:homotopy-terminal-for,def:equivariant-homotopy-equivalence}
\lean{BC4lean.ProperActions.homotopyTerminal_unique,BC4lean.ProperActions.homotopyTerminal_fixedPoints_unique,BC4lean.ProperActions.HomotopyTerminalFor.of_equiv}
Suppose the homotopy-terminal mapping condition holds for each of the
four ordered pairs $(X,X),(X,Y),(Y,X),(Y,Y)$. Then $X$ and $Y$ are
equivariantly homotopy equivalent, and their $H$-fixed-point spaces are
homotopy equivalent for every subgroup $H$. The mapping condition is
preserved when its target is replaced by an equivariantly homotopy-equivalent space.
These are conditional statements; they do not establish the mapping
condition for a general model satisfying the fixed-point criterion.
\end{proposition}
\begin{proof}
\leanok
Choose maps in both directions. Uniqueness up to homotopy for the two
endomorphism spaces identifies the composites with the identity maps.
For replacement of the target, compose with the equivalence and its
inverse and use their equivariant homotopies.
\end{proof}

\begin{proposition}\label{prop:finite-group-point-model}
\uses{def:proper-model-fixed-point-criterion,def:homotopy-terminal-for,prop:proper-action-examples}
\lean{BC4lean.ProperActions.Point.fixedPointCriterion,BC4lean.ProperActions.Point.homotopyTerminal,BC4lean.ProperActions.Point.proper}
For a finite discrete group, the trivial action on a point is proper and
satisfies the fixed-point criterion. For every $\Gamma$-space $X$, maps
from $X$ to this point exist and are unique, hence unique up to
equivariant homotopy. This verifies the topological and mapping claims
for the finite-group point model; a formal CW structure is not asserted here.
\end{proposition}
\begin{proof}
\leanok
Every subgroup is finite, and each fixed-point space is a point. A
finite discrete group acts properly on a locally compact Hausdorff point.
The constant map is the unique equivariant map to the point.
\end{proof}

\section{Cocompact indexing for a given ambient space}

\begin{proposition}\label{prop:invariant-subspace-orbit-embedding}
\uses{def:equivariant-map}
\lean{BC4lean.ProperActions.InvariantSubset.orbitInclusion_isEmbedding,BC4lean.ProperActions.InvariantSubset.compactSpace_orbit_iff}
For a continuous $\Gamma$-action on $E$ and an invariant subset $X$ with
its subspace topology, $X/\Gamma\to E/\Gamma$ is a topological embedding.
Thus $X/\Gamma$ is compact if and only if the image of $X$ in $E/\Gamma$
is compact. No closedness or properness is needed for this assertion.
\end{proposition}
\begin{proof}
\leanok
The orbit projection is an open quotient map. Apply Mathlib's
subquotient embedding theorem to the invariant inclusion $X\hookrightarrow E$.
\end{proof}

\begin{definition}\label{def:closed-cocompact-piece}
\uses{prop:invariant-subspace-orbit-embedding}
\lean{BC4lean.ProperActions.CocompactPiece}
\leanok
For a given $\Gamma$-space $E$, a closed cocompact piece is a closed
invariant subset $X\subseteq E$ whose image in $E/\Gamma$ is compact.
For continuous actions, this is equivalent to compactness of its
intrinsic quotient $X/\Gamma$. Pieces are ordered by inclusion.
\end{definition}

\begin{proposition}\label{prop:cocompact-pieces-directed}
\uses{def:closed-cocompact-piece}
\lean{BC4lean.ProperActions.CocompactPiece.empty,BC4lean.ProperActions.CocompactPiece.union,BC4lean.ProperActions.CocompactPiece.directed,BC4lean.ProperActions.CocompactPiece.compactSpace_orbit,BC4lean.ProperActions.CocompactPiece.locallyCompact}
The closed cocompact pieces form a nonempty directed indexing system:
the empty set is a piece, and $X\cup Y$ is a common upper bound.
For continuous actions every piece has compact intrinsic quotient.
If $E$ is locally compact, every piece is locally compact.
\end{proposition}
\begin{proof}
\leanok
A finite union of closed invariant sets is closed and invariant, and
its image is the finite union of the compact images. Closed subspaces
inherit local compactness.
\end{proof}

\begin{proposition}\label{prop:cocompact-pieces-cover}
\uses{def:closed-cocompact-piece,prop:proper-orbit-space}
\lean{BC4lean.ProperActions.CocompactPiece.ofCompact,BC4lean.ProperActions.CocompactPiece.subset_ofCompact,BC4lean.ProperActions.CocompactPiece.covers}
For a proper $\Gamma$-space $E$, every compact subset $K\subseteq E$
is contained in a closed cocompact invariant piece, namely its saturation
$q^{-1}(q(K))$, where $q:E\to E/\Gamma$. In particular the pieces cover $E$.
\end{proposition}
\begin{proof}
\leanok
The compact image $q(K)$ is closed in the Hausdorff orbit space. Its
preimage is closed and invariant, and its quotient image is exactly
$q(K)$. Apply this construction to singletons for the covering assertion.
\end{proof}

\section{Categorical equivariant cell attachments}

\begin{definition}\label{def:equivariant-cellular-data}
\uses{def:equivariant-map,def:orbit-cell}
\lean{BC4lean.ProperActions.topologicalAction,BC4lean.ProperActions.EquivariantMap.actionHomEquiv,BC4lean.ProperActions.equivariantBasicCell,BC4lean.ProperActions.RelativeEquivariantCWComplex,BC4lean.ProperActions.EquivariantCWComplex}
\leanok
In the category of topological $\Gamma$-actions, use the boundary inclusions
$\Gamma/H\times\partial D^n\to\Gamma/H\times D^n$, indexed by finite subgroups
$H$, as the generating cells at stage $n$. A relative equivariant CW structure
is a sequential relative cell complex for this family: its data contain the
coproducts, pushout squares, skeleta, and colimit witnesses. An equivariant CW
structure starts from the initial, empty action. The categorical morphisms
between actions obtained from $\Gamma$-spaces correspond exactly to the
continuous equivariant maps defined above.

These are cellular data. Topological properness of the resulting colimit is
not assumed in this definition and remains to be proved.
\end{definition}

\begin{proposition}\label{prop:orbit-boundary-cell-properties}
\uses{def:equivariant-cellular-data,prop:proper-orbit-cells}
\lean{BC4lean.ProperActions.OrbitCell.boundaryInclusion_injective,BC4lean.ProperActions.OrbitCell.disk_proper,BC4lean.ProperActions.OrbitCell.boundary_proper}
Every orbit boundary inclusion is injective. For finite $H$, both
$\Gamma/H\times D^n$ and $\Gamma/H\times\partial D^n$ are proper
$\Gamma$-spaces, including when $n=0$.
\end{proposition}
\begin{proof}
\leanok
The inclusion preserves the coset coordinate and is the usual injective
boundary inclusion in the second coordinate. The disk and its boundary are
compact Hausdorff, so the previously verified orbit-cell properness theorem
applies.
\end{proof}

\begin{proposition}\label{prop:equivariant-cellular-map-ext}
\uses{def:equivariant-cellular-data}
\lean{BC4lean.ProperActions.RelativeEquivariantCWComplex.hom_ext}
For a relative equivariant CW structure $X\to Y$, two equivariant continuous
maps $Y\to Z$ are equal if they agree on $X$ and on every characteristic cell
map into $Y$.
\end{proposition}
\begin{proof}
\leanok
Apply the morphism extensionality theorem for relative cell complexes, using
the pushout universal property at successor stages and the colimit universal
property for the resulting cellular object.
\end{proof}

\begin{proposition}\label{prop:orbit-cell-extension-reduction}
\uses{def:equivariant-cellular-data,prop:orbit-fixed-point-correspondence,def:subgroup-fixed-point-space}
\lean{BC4lean.ProperActions.orbitCellMapEquiv,BC4lean.ProperActions.cellEvaluation_comp,BC4lean.ProperActions.orbitCell_extension_iff}
Let $H$ be any subgroup and let the action on the parameter spaces $D,E$ be
trivial. For a target $\Gamma$-space $X$, evaluation on the identity-coset slice
gives a bijection between continuous equivariant maps
$\Gamma/H\times D\to X$ and continuous maps $D\to X^H$.
For a continuous map $i:D\to E$, a map on $\Gamma/H\times D$ extends across
$\mathrm{id}\times i$ if and only if its fixed-point slice extends across $i$.
This is a bijection of sets of maps, not a claimed homeomorphism of mapping spaces.
\end{proposition}
\begin{proof}
\leanok
The inverse sends $u:D\to X^H$ to $(gH,d)\mapsto g\,u(d)$.
Fixed-point membership makes it well-defined, and the discrete coset factor
reduces continuity to continuity on each slice. Evaluation and this construction
are inverse and commute with parameter restriction, giving the extension
criterion. Existence of the ordinary fixed-point extensions is not asserted here.
\end{proof}


\begin{proposition}\label{prop:contractible-disk-extension}
\lean{BC4lean.ProperActions.nullhomotopic_extends_ball,BC4lean.ProperActions.contractible_extends_disk}
Every continuous map $f:\partial D^n\to Y$ into a contractible space $Y$
extends to a continuous map $F:D^n\to Y$, including when $n=0$.
More generally, a null-homotopic map from the unit sphere of a proper real
normed vector space extends over its closed unit ball.
\end{proposition}
\begin{proof}
\leanok
For a nonempty unit sphere, the radial map $(t,s)\mapsto ts$ from its cylinder
onto the closed unit ball is a continuous surjection from a compact space to a
Hausdorff space, hence a quotient map. A reversed null-homotopy is constant
on its fibers: only the zero-radius fiber is collapsed. Descending this
homotopy gives the extension. For an empty sphere use the constant map at
the point supplied by the null-homotopy.
\end{proof}

\uses{prop:contractible-disk-extension,prop:orbit-cell-extension-reduction,def:proper-model-fixed-point-criterion}

\end{proposition}
\begin{proof}
\leanok
Evaluate on the identity-coset slice, extend the resulting map into $X^H$
over the disk, and apply the inverse orbit/fixed-point correspondence.
\end{proof}

\begin{proposition}\label{prop:equivariant-cellular-extension}
\uses{def:equivariant-cellular-data,prop:contractible-orbit-cell-extension}

\begin{proposition}\label{prop:disk-homotopy-extension}
\uses{prop:contractible-disk-extension}
\lean{BC4lean.ProperActions.contractible_extends_ball_homotopy,BC4lean.ProperActions.contractible_extends_disk_homotopy,BC4lean.ProperActions.contractible_disk_path_lifting}
Let $Y$ be contractible, let $f_0,f_1:D^n\to Y$ be continuous, and let
$H:[0,1]\times\partial D^n\to Y$ be a homotopy between their restrictions.
There is a homotopy $F:[0,1]\times D^n\to Y$ from $f_0$ to $f_1$ satisfying
$F(t,s)=H(t,s)$ on the whole boundary, including when $n=0$.
Equivalently, every family of boundary paths with prescribed disk endpoints
extends to a continuous family of disk paths, with those endpoints unchanged.
\end{proposition}
\begin{proof}
\leanok
Use the max norm on $\mathbb R\times\mathbb R^n$. Its unit sphere is a quotient
of the disjoint union of the two disk faces and the cylindrical side.
The endpoint conditions make $f_0,f_1,H$ agree on the fibers of this quotient.
Descend them to a boundary map and apply the verified ball extension theorem.
Parametrizing the resulting max-norm ball by $(t,x)\mapsto(2t-1,x)$ gives
$F$. The path-family formulation follows by currying; the unit interval is
locally compact, so evaluation is jointly continuous.
\end{proof}

\begin{proof}
\leanok

coset, apply disk path lifting in $X^H$, and recover the equivariant family

\end{proof}

\section{A concrete probability target}

\begin{definition}\label{def:probability-simplex}
\lean{BC4lean.ProperActions.probabilitySet,BC4lean.ProperActions.ProbabilitySimplex,BC4lean.ProperActions.ProbabilitySimplex.instMulAction,BC4lean.ProperActions.ProbabilitySimplex.instContinuousConstSMul}
\leanok
Let $P(\Gamma)$ be the space of functions $p:\Gamma\to\mathbb R$ such that
$p(g)\geq0$, $p$ is summable, and $\sum_g p(g)=1$. Give it the subspace
topology of pointwise convergence in $\mathbb R^\Gamma$ and the action
$(a\cdot p)(g)=p(a^{-1}g)$. Each translation is continuous.
This definition supplies a topological action; it does not supply a CW
structure or local compactness.
\end{definition}

\begin{proposition}\label{prop:probability-fixed-points}
\uses{def:probability-simplex,def:subgroup-fixed-point-space,def:proper-model-fixed-point-criterion}
\lean{BC4lean.ProperActions.probabilityFixedPointsHomeomorph,BC4lean.ProperActions.ProbabilitySimplex.fixedPoints_contractible,BC4lean.ProperActions.ProbabilitySimplex.fixedPoints_empty,BC4lean.ProperActions.ProbabilitySimplex.fixedPointCriterion}
For finite $H\leq\Gamma$, $P(\Gamma)^H$ is contractible. For infinite
$H\leq\Gamma$, it is empty. Thus $P(\Gamma)$ satisfies the fixed-point criterion.
\end{proposition}
\begin{proof}
\leanok
The fixed-point space is homeomorphic to the convex set of invariant
probability functions. For finite $H$, this set contains the uniform
probability function on $H$. For infinite $H$, a nonzero nonnegative invariant
function has a positive constant value on an infinite coset, contradicting
summability. The total mass condition excludes the zero function.
\end{proof}

\begin{proposition}\label{prop:probability-proper-action}
\uses{def:probability-simplex,def:proper-action}
\lean{BC4lean.ProperActions.ProbabilitySimplex.compact_half_mass,BC4lean.ProperActions.ProbabilitySimplex.finite_compact_transporter,BC4lean.ProperActions.ProbabilitySimplex.proper,BC4lean.ProperActions.ProbabilitySimplex.secondCountable}
For countable discrete $\Gamma$, $P(\Gamma)$ is a proper second-countable
$\Gamma$-space. More generally, compact subsets of $P(\Gamma)$ have finite
transporters for every discrete group.
\end{proposition}
\begin{proof}
\leanok
Compactness gives a finite set carrying more than half the mass of every
probability function in a given compact subset. If $gK\cap L\ne\varnothing$
and $S,T$ are such finite sets for $K,L$, respectively, then $gS\cap T$
is nonempty; hence $g\in TS^{-1}$. The transporter is finite.
For countable $\Gamma$, the function space and its subspace are first
countable and Hausdorff, so the compact-transporter criterion implies
properness. Second countability follows from the countable product topology.
\end{proof}

\begin{proposition}\label{prop:properness-equivariant-pullback}
\uses{def:proper-action,def:equivariant-map}
\lean{BC4lean.ProperActions.EquivariantMap.properSMul}
Let $A$ be a Hausdorff continuous $\Gamma$-space, let $X$ be a proper
$\Gamma$-space, and let $f:A\to X$ be continuous and equivariant.
Then the action on $A$ is proper. No properness assumption on the map $f$
is needed.
\end{proposition}
\begin{proof}
\leanok
Push an ultrafilter on $\Gamma\times A$ forward along
$(g,a)\mapsto(g,f(a))$. Properness of the action on $X$ supplies convergence
of its group coordinate. The Hausdorff ultrafilter criterion for the action
on $A$ then proves properness.
\end{proof}

\begin{proposition}\label{prop:probability-cellular-target}
\uses{prop:probability-fixed-points,prop:probability-proper-action,prop:equivariant-cellular-homotopy-terminal}
\lean{BC4lean.ProperActions.EquivariantCWComplex.probability_homotopyTerminal,BC4lean.ProperActions.probability_proper_universal_target}
Every finite-isotropy equivariant CW complex admits a map to $P(\Gamma)$,

$P(\Gamma)$ has not been constructed.
\end{proposition}
\begin{proof}
\leanok
Apply the verified cellular mapping theorem to the explicit target and its
fixed-point criterion. Its properness is given by the preceding proposition.
\end{proof}

\begin{proposition}\label{prop:hausdorff-equivariant-cw-proper}
\uses{prop:probability-cellular-target,prop:properness-equivariant-pullback}
\lean{BC4lean.ProperActions.EquivariantCWComplex.proper_of_t2}

\section{Closed skeletal inclusions and the colimit topology}

\begin{proposition}\label{prop:orbit-boundary-closed-embedding}
\uses{def:orbit-cell,prop:orbit-boundary-cell-properties}
\lean{BC4lean.ProperActions.OrbitCell.boundaryInclusion_isClosedEmbedding}
\leanok
For every subgroup $H\leq\Gamma$ and every $n\geq0$, the inclusion
$\Gamma/H\times\partial D^n\longrightarrow\Gamma/H\times D^n$ is a closed embedding.
\end{proposition}
\begin{proof}
\leanok
The disk boundary is compact and the disk is Hausdorff, so the boundary
inclusion is a closed embedding. Take its product with the identity on
$\Gamma/H$. Finiteness of $H$ is not required here.
\end{proof}

\begin{proposition}\label{prop:cellular-closed-inclusions}
\uses{def:equivariant-cellular-data,prop:orbit-boundary-closed-embedding}
\lean{BC4lean.ProperActions.orbitCellAttachment_isClosedEmbedding,BC4lean.ProperActions.RelativeEquivariantCWComplex.skeletonMap_isClosedEmbedding,BC4lean.ProperActions.RelativeEquivariantCWComplex.skeletonInclusion_isClosedEmbedding,BC4lean.ProperActions.RelativeEquivariantCWComplex.isClosedEmbedding}
\leanok
Let $A\longrightarrow X$ carry a relative equivariant CW structure, with
skeletal sequence $(X_n)_{n\geq0}$. The map $A\longrightarrow X$, every
transition $X_m\longrightarrow X_n$ for $m\leq n$, and every inclusion
$X_n\longrightarrow X$ are closed embeddings. In particular, no distinct
points of a skeleton are identified in later stages or in the colimit.
\end{proposition}
\begin{proof}
\leanok
Coproducts and pushouts preserve closed embeddings in the category of
topological spaces. This proves the assertion for successive attachments,
and composition gives all finite transitions. The underlying set functor
preserves colimits; stability of injective maps under these cellular
colimits proves injectivity of the skeletal inclusions. For a closed subset
of one skeleton, its inverse image on an earlier skeleton is a continuous
preimage, and on a later skeleton is a closed image. The colimit topology
therefore makes its image closed in $X$.
\end{proof}

\begin{proposition}\label{prop:cellular-skeletal-topology}
\uses{def:equivariant-cellular-data}
\lean{BC4lean.ProperActions.RelativeEquivariantCWComplex.exists_skeleton_preimage,BC4lean.ProperActions.RelativeEquivariantCWComplex.isOpen_iff,BC4lean.ProperActions.RelativeEquivariantCWComplex.isClosed_iff,BC4lean.ProperActions.RelativeEquivariantCWComplex.continuous_iff}
\leanok
Every point of $X$ comes from some finite skeletal stage. A subset of $X$
is open, respectively closed, exactly when its inverse image in every
$X_n$ is open, respectively closed. A function $X\longrightarrow Z$ into
any topological space is continuous exactly when its restriction to every
skeletal stage is continuous.
\end{proposition}
\begin{proof}
\leanok
Forget the action and use the corresponding set and topological colimit
universal properties.
\end{proof}
