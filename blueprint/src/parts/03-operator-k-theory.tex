\chapter{Operator K-theory and the analytic side}
\label{ch:operator-k-theory}

Unless stated otherwise, $A$ is a unital complex $C^*$-algebra.
The present Lean milestone constructs the stable projection monoid $V(A)$
and the unital group $K_0(A)$ with its universal property and functoriality
under unital complex star homomorphisms, including canonical additive
isomorphisms induced by unital star-algebra isomorphisms. Matrix unitaries,
fixed-size norm homotopies, stabilization maps, and explicit block-rotation
homotopies giving the stable inverse mechanism are also verified.
The stable group $K_1$, nonunital extensions, and later structural properties
remain separate targets.

\section{Projections and Murray--von Neumann equivalence}

\begin{definition}\label{def:operator-projection}
\lean{BC4lean.OperatorKTheory.Projection,BC4lean.OperatorKTheory.Projection.zero,BC4lean.OperatorKTheory.Projection.one,BC4lean.OperatorKTheory.Projection.complement}
\leanok
A projection in $A$ is an element $p$ satisfying $p^*=p$ and $p^2=p$.
Write $\operatorname{Proj}(A)$ for these projections. The elements $0$, $1$,
and $1-p$ are projections.
\end{definition}

\begin{proposition}\label{prop:partial-isometry-support}
\uses{def:operator-projection}
\lean{BC4lean.OperatorKTheory.Projection.right_support}
If $p\in\operatorname{Proj}(A)$ and $v^*v=p$, then $vp=v$.
\end{proposition}
\begin{proof}
\leanok
For $x=vp-v$, expansion using $p^*=p$, $p^2=p$, and $v^*v=p$
gives $x^*x=0$. The $C^*$-identity implies $x=0$.
\end{proof}

\begin{definition}\label{def:projection-equivalence}
\uses{def:operator-projection}
\lean{BC4lean.OperatorKTheory.Projection.Equivalent}
\leanok
Projections $p,q\in A$ are Murray--von Neumann equivalent, written $p\sim q$,
if some $v\in A$ satisfies $v^*v=p$ and $vv^*=q$.
\end{definition}

\begin{proposition}\label{prop:projection-equivalence}
\uses{def:projection-equivalence,prop:partial-isometry-support}
\lean{BC4lean.OperatorKTheory.Projection.equivalent_refl,BC4lean.OperatorKTheory.Projection.equivalent_symm,BC4lean.OperatorKTheory.Projection.equivalent_trans,BC4lean.OperatorKTheory.Projection.equivalenceSetoid,BC4lean.OperatorKTheory.Projection.Classes}
The relation $\sim$ is an equivalence relation, so its quotient is defined.
\end{proposition}
\begin{proof}
\leanok
Reflexivity uses $v=p$ and symmetry uses $v^*$. If $v$ implements $p\sim q$
and $w$ implements $q\sim r$, then $wv$ implements $p\sim r$.
The support identities $vp=v$, $qv=v$, and $wq=w$ give the two required equations.
These quotients concern one algebra at a time; they are not yet stable classes.
\end{proof}

\begin{proposition}\label{prop:projection-map}
\uses{prop:projection-equivalence}
\lean{BC4lean.OperatorKTheory.Projection.map,BC4lean.OperatorKTheory.Projection.equivalent_map,BC4lean.OperatorKTheory.Projection.mapClasses}
A unital complex star homomorphism $\phi:A\to B$ sends projections to
projections and preserves Murray--von Neumann equivalence. It therefore
induces a map on the preceding quotients.
\end{proposition}
\begin{proof}
\leanok
Apply $\phi$ to the projection equations and, for equivalence, to the
witness $v$ and its two defining equations.
\end{proof}

\section{Matrix stabilization and the construction of $K_0$}

\begin{definition}\label{def:cstar-matrix-projection}
\uses{def:operator-projection}
\lean{BC4lean.OperatorKTheory.MatrixProjection,BC4lean.OperatorKTheory.matrixBlockSum,BC4lean.OperatorKTheory.projectionBlockSum,BC4lean.OperatorKTheory.stabilizeProjection}
\leanok
Use the $C^*$-norm on $M_n(A)$, and let $P_n(A)=\operatorname{Proj}(M_n(A))$.
For $p\in P_n(A)$ and $q\in P_m(A)$, their block sum is
\[
 p\oplus q=\begin{pmatrix}p&0\\0&q\end{pmatrix}\in P_{n+m}(A).
\]
Projection stabilization is $p\mapsto p\oplus 0_m$.
The Lean implementation initially uses arbitrary finite index types and their
disjoint sums. Mathlib's matrix norm requires a compatible star order on $A$;
the matrix declarations retain this explicit implementation assumption.
\end{definition}

\begin{proposition}\label{prop:projection-block-sum}
\uses{def:cstar-matrix-projection,prop:projection-equivalence}
\lean{BC4lean.OperatorKTheory.equivalent_blockSum,BC4lean.OperatorKTheory.equivalent_stabilize,BC4lean.OperatorKTheory.blockSumClasses}
If $p\sim p'$ and $q\sim q'$, then $p\oplus q\sim p'\oplus q'$.
Thus block sum and stabilization descend to equivalence classes at the
indicated matrix sizes.
\end{proposition}
\begin{proof}
\leanok
If $v$ and $w$ are the respective witnesses, use $v\oplus w$.
Block multiplication and adjoints give both defining equations.
\end{proof}

\begin{definition}\label{def:rectangular-projection-equivalence}
\uses{def:cstar-matrix-projection}
\lean{BC4lean.OperatorKTheory.projectionMatrix,BC4lean.OperatorKTheory.RectEquivalent}
\leanok
For finite index sets $I,J$, projections $p\in M_I(A)$ and $q\in M_J(A)$
are rectangularly equivalent if there is $v\in M_{I,J}(A)$ with
\[
 vv^*=p,\qquad v^*v=q.
\]
Unlike the earlier relation in one fixed algebra, this relation can compare
different matrix sizes directly.
\end{definition}

\begin{proposition}\label{prop:rectangular-equivalence-stable}
\uses{def:rectangular-projection-equivalence,prop:partial-isometry-support,prop:projection-block-sum}
\lean{BC4lean.OperatorKTheory.rectangular_right_support,BC4lean.OperatorKTheory.rectangular_left_support,BC4lean.OperatorKTheory.rectEquivalent_refl,BC4lean.OperatorKTheory.rectEquivalent_symm,BC4lean.OperatorKTheory.rectEquivalent_trans,BC4lean.OperatorKTheory.rectEquivalent_stabilize,BC4lean.OperatorKTheory.rectEquivalent_of_reindex,BC4lean.OperatorKTheory.rectEquivalent_iff_padded}
Rectangular equivalence is an equivalence relation on finite matrix projections
of varying sizes. It is unchanged by reindexing and zero-padding. Moreover,
\[
 p\sim_{\mathrm{rect}}q
 \quad\Longleftrightarrow\quad
 p\oplus0_J\sim 0_I\oplus q\quad\text{in }M_{I\sqcup J}(A),
\]
where the relation on the right is ordinary Murray--von Neumann equivalence.
Thus rectangular witnesses give a model for stable projection equivalence.
\end{proposition}
\begin{proof}
\leanok
Embed a rectangular matrix as an off-diagonal block in $M_{I\sqcup J}(A)$.
The square-matrix support lemma gives $vq=v$ from $v^*v=q$ and, by taking
adjoints, $pv=v$ from $vv^*=p$. These identities prove transitivity using
the product of rectangular witnesses. Reindexing uses submatrices along
bijections. Zero projections of any sizes are equivalent using the zero
rectangular matrix; block sum and reindexing then give zero-padding invariance.
Combine these equivalences with the fixed-size equivalence relation to obtain
the displayed common-corner characterization.
\end{proof}

\begin{definition}\label{def:stable-projection-monoid}
\uses{prop:rectangular-equivalence-stable}
\lean{BC4lean.OperatorKTheory.FiniteProjection,BC4lean.OperatorKTheory.FiniteProjection.Equivalent,BC4lean.OperatorKTheory.FiniteProjection.setoid,BC4lean.OperatorKTheory.StableProjectionMonoid,BC4lean.OperatorKTheory.StableProjectionMonoid.of,BC4lean.OperatorKTheory.StableProjectionMonoid.of_eq_iff}
\leanok
Let $V(A)$ be finite matrix projections modulo rectangular equivalence.
The Lean objects record a small finite index type, its finite structure,
and a projection. The relation is the existence of an actual rectangular
partial-isometry witness; block-sum relations are not imposed separately.
\end{definition}

\begin{proposition}\label{prop:stable-projection-monoid-laws}
\uses{def:stable-projection-monoid,prop:projection-block-sum}
\lean{BC4lean.OperatorKTheory.rectEquivalent_blockSum,BC4lean.OperatorKTheory.FiniteProjection.blockSum_congr,BC4lean.OperatorKTheory.FiniteProjection.blockSum_assoc,BC4lean.OperatorKTheory.FiniteProjection.blockSum_comm,BC4lean.OperatorKTheory.FiniteProjection.zero_blockSum,BC4lean.OperatorKTheory.StableProjectionMonoid.instAddCommMonoid,BC4lean.OperatorKTheory.StableProjectionMonoid.of_zeroProjection,BC4lean.OperatorKTheory.StableProjectionMonoid.of_stabilize}
Block sum makes $V(A)$ an additive commutative monoid. Any zero projection
represents its zero, and $[p\oplus0]=[p]$.
\end{proposition}
\begin{proof}
\leanok
Block sums of rectangular witnesses prove independence of representatives.
The associativity and block-switching bijections of finite index sets prove
associativity and commutativity in the quotient. The empty matrix gives the
identity. Zero-padding invariance follows from rectangular equivalence.
\end{proof}

\begin{proposition}\label{prop:stable-projection-finite-representative}
\uses{def:stable-projection-monoid}
\lean{BC4lean.OperatorKTheory.FiniteProjection.toFin,BC4lean.OperatorKTheory.FiniteProjection.equivalent_toFin,BC4lean.OperatorKTheory.StableProjectionMonoid.exists_finRepresentative}
Every class in $V(A)$ has a representative indexed by
$\{0,\ldots,n-1\}$ for some $n\geq0$.
\end{proposition}
\begin{proof}
\leanok
Reindex a projection using the bijection from its finite index type to the
standard finite ordinal of the same cardinality. Reindexing preserves its class.
\end{proof}

\begin{definition}\label{def:operator-k0}
\uses{prop:stable-projection-monoid-laws}
\lean{BC4lean.OperatorKTheory.K0,BC4lean.OperatorKTheory.k0Class,BC4lean.OperatorKTheory.k0Projection,BC4lean.OperatorKTheory.k0Projection_zero,BC4lean.OperatorKTheory.k0Projection_blockSum,BC4lean.OperatorKTheory.k0Projection_eq_of_equivalent}
\leanok
Define the unital group $K_0(A)$ as the Grothendieck group of $V(A)$.
The canonical additive map $V(A)\to K_0(A)$ sends a matrix projection to
its $K_0$ class and sends block sum to addition.
Cancellation in $V(A)$ is not assumed, and injectivity of this canonical map
is not asserted.
\end{definition}

\begin{proposition}\label{prop:operator-k0-universal}
\uses{def:operator-k0}
\lean{BC4lean.OperatorKTheory.k0Universal,BC4lean.OperatorKTheory.k0Universal_comp,BC4lean.OperatorKTheory.k0Universal_class,BC4lean.OperatorKTheory.k0Universal_unique}
For every abelian group $G$, composition with the canonical map induces a
bijection
\[
 \operatorname{Hom}_{\mathrm{Ab}}(K_0(A),G)
 \simeq \operatorname{Hom}_{\mathrm{Mon}}(V(A),G).
\]
\end{proposition}
\begin{proof}
\leanok
Apply the universal property of Mathlib's additive Grothendieck group to
the verified commutative monoid $V(A)$. The resulting extension agrees on
projection classes and is unique.
\end{proof}

\section{Functoriality of unital $K_0$}

\begin{proposition}\label{prop:matrix-projection-map}
\uses{prop:projection-map,prop:rectangular-equivalence-stable,prop:projection-block-sum}
\lean{BC4lean.OperatorKTheory.matrixMap,BC4lean.OperatorKTheory.matrixProjectionMap,BC4lean.OperatorKTheory.projectionMatrix_map,BC4lean.OperatorKTheory.matrixProjectionMap_blockSum,BC4lean.OperatorKTheory.rectEquivalent_map,BC4lean.OperatorKTheory.FiniteProjection.map,BC4lean.OperatorKTheory.FiniteProjection.equivalent_map}
Let $\phi:A\to B$ be a unital complex star homomorphism. Applying $\phi$
entrywise sends finite matrix projections over $A$ to projections over $B$,
preserves block sums and zero projections, and preserves rectangular equivalence.
The finite index types are unchanged.
\end{proposition}
\begin{proof}
\leanok
The entrywise map commutes with matrix multiplication and adjoints.
If $vv^*=p$ and $v^*v=q$, then the entrywise image of $v$ witnesses the
corresponding equations for the images of $p$ and $q$. Zero off-diagonal
blocks remain zero, so block sums are preserved. No injectivity of $\phi$
is required.
\end{proof}

\begin{definition}\label{def:stable-projection-map}
\uses{prop:matrix-projection-map,prop:stable-projection-monoid-laws}
\lean{BC4lean.OperatorKTheory.StableProjectionMonoid.map,BC4lean.OperatorKTheory.StableProjectionMonoid.map_of,BC4lean.OperatorKTheory.StableProjectionMonoid.map_id,BC4lean.OperatorKTheory.StableProjectionMonoid.map_comp}
\leanok
The entrywise map descends to an additive map
\[
 V(\phi):V(A)\longrightarrow V(B),\qquad [p]\longmapsto[\phi(p)].
\]
These maps preserve identities and composition.
\end{definition}

\begin{definition}\label{def:unital-k0-map}
\uses{def:stable-projection-map,prop:operator-k0-universal}
\lean{BC4lean.OperatorKTheory.k0Map,BC4lean.OperatorKTheory.k0Map_class,BC4lean.OperatorKTheory.k0Map_comp_class,BC4lean.OperatorKTheory.k0Map_projection}
\leanok
The universal property of the Grothendieck group extends $V(\phi)$ to
\[
 K_0(\phi):K_0(A)\longrightarrow K_0(B),\qquad
 K_0(\phi)([p])=[\phi(p)].
\]
Here $[p]$ denotes the $K_0$ class of a finite matrix projection, and
$\phi(p)$ denotes its entrywise image. The canonical maps $V(A)\to K_0(A)$
are natural with respect to these homomorphisms.
\end{definition}

\begin{proposition}\label{prop:unital-k0-functoriality}
\uses{def:unital-k0-map,prop:operator-k0-universal}
\lean{BC4lean.OperatorKTheory.k0Hom_ext,BC4lean.OperatorKTheory.k0Map_unique,BC4lean.OperatorKTheory.k0Map_id,BC4lean.OperatorKTheory.k0Map_comp}
For unital complex star homomorphisms $\phi:A\to B$ and $\psi:B\to C$,
\[
 K_0(\mathrm{id}_A)=\mathrm{id}_{K_0(A)},\qquad
 K_0(\psi\circ\phi)=K_0(\psi)\circ K_0(\phi).
\]
Moreover, $K_0(\phi)$ is the unique group homomorphism satisfying the
projection-class formula above.
\end{proposition}
\begin{proof}
\leanok
The universal property implies that homomorphisms out of $K_0(A)$ are
determined by their values on projection classes. Evaluate both sides of
each asserted equality on such a class and use the identity and composition
laws of the entrywise maps. This proof does not assume cancellation in $V(A)$.
\end{proof}

\section{Invariance under unital star-algebra isomorphism}

\begin{proposition}\label{prop:stable-projection-isomorphism}
\uses{def:stable-projection-map}
\lean{BC4lean.OperatorKTheory.StableProjectionMonoid.map_leftInverse,BC4lean.OperatorKTheory.StableProjectionMonoid.map_rightInverse,BC4lean.OperatorKTheory.StableProjectionMonoid.mapEquiv,BC4lean.OperatorKTheory.StableProjectionMonoid.mapEquiv_of,BC4lean.OperatorKTheory.StableProjectionMonoid.mapEquiv_refl,BC4lean.OperatorKTheory.StableProjectionMonoid.mapEquiv_symm,BC4lean.OperatorKTheory.StableProjectionMonoid.mapEquiv_trans}
A unital complex star-algebra isomorphism $e:A\to B$ induces an additive
isomorphism $V(e):V(A)\to V(B)$ sending $[p]$ to $[e(p)]$.
Its inverse is $V(e^{-1})$. This construction preserves identity
isomorphisms and composition and takes inverses to inverses.
\end{proposition}
\begin{proof}
\leanok
Apply the identity and composition laws for $V$ to $e^{-1}\circ e=\mathrm{id}_A$
and $e\circ e^{-1}=\mathrm{id}_B$. The two induced additive maps are mutual
inverses. The projection-class formula and coherence laws follow from those
for the entrywise maps.
\end{proof}

\begin{proposition}\label{prop:unital-k0-isomorphism}
\uses{prop:unital-k0-functoriality,prop:stable-projection-isomorphism}
\lean{BC4lean.OperatorKTheory.k0Map_leftInverse,BC4lean.OperatorKTheory.k0Map_rightInverse,BC4lean.OperatorKTheory.k0Equiv,BC4lean.OperatorKTheory.k0Equiv_toAddMonoidHom,BC4lean.OperatorKTheory.k0Equiv_class,BC4lean.OperatorKTheory.k0Equiv_projection,BC4lean.OperatorKTheory.k0Equiv_refl,BC4lean.OperatorKTheory.k0Equiv_symm,BC4lean.OperatorKTheory.k0Equiv_trans}
A unital complex star-algebra isomorphism $e:A\to B$ induces a canonical
additive isomorphism
\[
 K_0(e):K_0(A)\xrightarrow{\;\cong\;}K_0(B),\qquad [p]\longmapsto[e(p)],
\]
with inverse $K_0(e^{-1})$. The induced isomorphisms preserve identities
and composition, and the group-completion maps $V(A)\to K_0(A)$ commute
with these isomorphisms.
\end{proposition}
\begin{proof}
\leanok
The verified functorial laws give
$K_0(e^{-1})K_0(e)=\mathrm{id}_{K_0(A)}$ and
$K_0(e)K_0(e^{-1})=\mathrm{id}_{K_0(B)}$.
These two identities supply the inverse data for a bundled additive equivalence.
The class formula and compatibility with identity, inverse, and composition
follow from the corresponding laws for $K_0$ homomorphisms.
\end{proof}

\section{Unitaries, norm homotopy, and $K_1$}

\begin{definition}\label{def:matrix-unitary}
\lean{BC4lean.OperatorKTheory.MatrixUnitary,BC4lean.OperatorKTheory.unitaryBlockSum,BC4lean.OperatorKTheory.stabilizeUnitary}
\leanok
Let $U_n(A)=\{u\in M_n(A):u^*u=uu^*=1\}$, with the subspace topology
from the matrix $C^*$-norm. Block sums of unitaries are unitary.
Unitary stabilization is $u\mapsto u\oplus 1_m$.
\end{definition}

\begin{definition}\label{def:unitary-norm-homotopy}
\uses{def:matrix-unitary}
\lean{BC4lean.OperatorKTheory.UnitaryHomotopic,BC4lean.OperatorKTheory.UnitaryHomotopyClasses}
\leanok
Two elements of $U_n(A)$ are norm homotopic if they are endpoints of a
continuous path $[0,1]\to U_n(A)$. Form the quotient by this equivalence
relation at each fixed size.
\end{definition}

\begin{proposition}\label{prop:unitary-homotopy-operations}
\uses{def:unitary-norm-homotopy}
\lean{BC4lean.OperatorKTheory.unitaryHomotopic_refl,BC4lean.OperatorKTheory.unitaryHomotopic_symm,BC4lean.OperatorKTheory.unitaryHomotopic_trans,BC4lean.OperatorKTheory.unitaryHomotopic_mul,BC4lean.OperatorKTheory.unitaryHomotopic_inv}
Norm homotopy is an equivalence relation and is preserved by multiplication
of pairs of unitaries and by inversion.
\end{proposition}
\begin{proof}
\leanok
Use constant paths, reversal, concatenation, pointwise multiplication,
and pointwise inversion in the topological unitary group.
\end{proof}

\begin{proposition}\label{prop:unitary-homotopy-stabilization}
\uses{prop:unitary-homotopy-operations,def:matrix-unitary}
\lean{BC4lean.OperatorKTheory.continuous_matrixBlockSum,BC4lean.OperatorKTheory.continuous_unitaryBlockSum,BC4lean.OperatorKTheory.unitaryHomotopic_blockSum,BC4lean.OperatorKTheory.unitaryHomotopic_stabilize,BC4lean.OperatorKTheory.stabilizeUnitaryClasses}
Block sum is continuous and preserves norm homotopies. Consequently,
stabilization induces maps on the fixed-size norm-path components.
\end{proposition}
\begin{proof}
\leanok
The finite matrix topology agrees with the entrywise product topology,
in which block sum is continuous. Apply block sum to a pair of paths;
for stabilization, use a constant identity path in the second factor.
\end{proof}

\begin{proposition}\label{prop:unitary-block-rotation}
\uses{prop:unitary-homotopy-stabilization}
\lean{BC4lean.OperatorKTheory.rotationMatrix,BC4lean.OperatorKTheory.rotationMatrix_mem_unitary,BC4lean.OperatorKTheory.rotationUnitary,BC4lean.OperatorKTheory.continuous_rotationUnitary,BC4lean.OperatorKTheory.rotationUnitary_zero,BC4lean.OperatorKTheory.rotationUnitary_conjugate,BC4lean.OperatorKTheory.unitaryBlockProductPath,BC4lean.OperatorKTheory.unitaryHomotopic_blockSum_mul}
For $u,v\in U_n(A)$, there is an explicit norm-continuous path
\[
 u\oplus v\simeq (uv)\oplus 1_n\quad\text{in }U_{2n}(A).
\]
\end{proposition}
\begin{proof}
\leanok
Put $\theta=\pi t/2$ for $0\leq t\leq1$ and let
\[
 R(\theta)=\begin{pmatrix}
 \cos\theta\,1_n&-\sin\theta\,1_n\\
 \sin\theta\,1_n&\cos\theta\,1_n
 \end{pmatrix}.
\]
The identity $\cos^2\theta+\sin^2\theta=1$ proves that $R(\theta)$ is
unitary. Its entries depend continuously on $\theta$, hence it is continuous
in the finite-matrix $C^*$-norm. The required path is
\[
 H_{u,v}(t)=(u\oplus1_n)R(\theta)(1_n\oplus v)R(\theta)^*.
\]
At $t=0$ the rotation is the identity. At $t=1$ its conjugation exchanges
the two diagonal blocks, giving $(uv)\oplus1_n$.
\end{proof}

\begin{proposition}\label{prop:unitary-stable-inverse}
\uses{prop:unitary-block-rotation}
\lean{BC4lean.OperatorKTheory.unitaryInversePath,BC4lean.OperatorKTheory.unitaryHomotopic_blockSum_star,BC4lean.OperatorKTheory.unitaryHomotopic_star_blockSum}
For every $u\in U_n(A)$, both $u\oplus u^*$ and $u^*\oplus u$ are
norm homotopic to $1_{2n}$.
\end{proposition}
\begin{proof}
\leanok
Specialize the preceding path to $v=u^*$ and use $uu^*=1_n$.
Apply the same result to $u^*$ for the second assertion.
This proves the inverse homotopy needed for the stable quotient; it does
not by itself construct that quotient or its abelian group structure.
\end{proof}

\begin{definition}\label{def:operator-k1}
\uses{prop:unitary-homotopy-stabilization,prop:unitary-stable-inverse}
\notready
Take the stable quotient of matrix unitaries by norm homotopy after adjoining
identity blocks. Define $K_1(A)$ to be this quotient and prove that block sum
gives an abelian group. This requires the stabilization and reindexing
coherence and block-switching homotopies. The preceding result supplies
the homotopy needed for an inverse represented by $u^*$, but the stable
quotient and its abelian group laws remain to be constructed.
\end{definition}

\section{The analytic target}

\begin{definition}\label{def:analytic-k0}
\uses{def:reduced-group-cstar,def:operator-k0,prop:operator-k0-universal}
\lean{BC4lean.ReducedProjectionMonoid,BC4lean.reducedProjectionAddCommMonoid,BC4lean.ReducedK0,BC4lean.reducedK0AddCommGroup,BC4lean.reducedK0Class,BC4lean.reducedK0Universal}
\leanok
For a discrete group $\Gamma$, equip the already constructed algebra
$C_r^*(\Gamma)$ with its canonical spectral order. Define the abelian group
$K_0(C_r^*(\Gamma))$ by the preceding unital construction, together with
its projection-class map and universal property. Countability is not needed
for this construction. No computation of this group or assembly map is
asserted here.
\end{definition}


\begin{definition}\label{def:abstract-analytic-k0}
\uses{prop:reduced-group-cstar-completion,def:operator-k0}
\lean{BC4lean.ReducedCompletionProjectionMonoid,BC4lean.reducedCompletionProjectionAddCommMonoid,BC4lean.ReducedCompletionK0,BC4lean.reducedCompletionK0AddCommGroup,BC4lean.reducedCompletionK0Class}
\leanok
For a group $\Gamma$, regarded as discrete, write
\[
 A_r(\Gamma)=\widehat{\mathbb C[\Gamma]}^{\|\cdot\|_r}
\]
for the actual abstract reduced-norm completion constructed earlier.
Equip this $C^*$-algebra with its canonical spectral order and form
$V(A_r(\Gamma))$, $K_0(A_r(\Gamma))$, and the canonical group-completion
map $\iota_A:V(A_r(\Gamma))\to K_0(A_r(\Gamma))$.
These definitions use the existing matrix-projection and Grothendieck-group
constructions applied to the abstract completion itself.
\end{definition}

\begin{proposition}\label{prop:abstract-concrete-k0}
\uses{def:abstract-analytic-k0,def:analytic-k0,prop:unital-k0-isomorphism,prop:reduced-group-cstar-completion}
\lean{BC4lean.reducedCompletionProjectionMap,BC4lean.reducedCompletionProjectionMap_entry,BC4lean.reducedCompletionProjectionEquiv,BC4lean.reducedCompletionProjectionEquiv_of,BC4lean.reducedCompletionProjectionEquiv_symm_of,BC4lean.reducedCompletionK0Equiv,BC4lean.reducedCompletionK0Equiv_class,BC4lean.reducedCompletionK0Equiv_symm_class,BC4lean.reducedCompletionK0Equiv_comp_class,BC4lean.reducedCompletionK0Equiv_projection,BC4lean.reducedCompletionK0Equiv_unique}
The canonical star-algebra isomorphism
$e_\Gamma:A_r(\Gamma)\to C_r^*(\Gamma)$ induces additive isomorphisms
\[
 V(e_\Gamma):V(A_r(\Gamma))\xrightarrow{\;\cong\;}V(C_r^*(\Gamma)),
 \qquad
 K_0(e_\Gamma):K_0(A_r(\Gamma))\xrightarrow{\;\cong\;}K_0(C_r^*(\Gamma)).
\]
For a finite matrix projection $p$ over $A_r(\Gamma)$, its image is represented
by the matrix obtained by applying $e_\Gamma$ to every entry. In particular,
$K_0(e_\Gamma)([p])=[e_\Gamma(p)]$.
The inverse maps are induced by $e_\Gamma^{-1}$, and
\[
 K_0(e_\Gamma)\circ\iota_A=\iota_C\circ V(e_\Gamma),
\]
where $\iota_C:V(C_r^*(\Gamma))\to K_0(C_r^*(\Gamma))$ is the concrete
model's group-completion map. This equation uniquely determines the underlying
homomorphism $K_0(e_\Gamma)$. Countability of $\Gamma$ is not required.
\end{proposition}
\begin{proof}
\leanok
Apply the verified isomorphism-invariance constructions to the previously
verified star-algebra equivalence $e_\Gamma$, choosing the canonical spectral
order on both algebras. The entrywise and projection-class formulas specialize
the general formulas. Naturality gives the displayed commuting equation, and
extensionality on projection classes proves uniqueness.
This identifies two models of the analytic target; it does not compute that
target or construct the Baum--Connes assembly map.
\end{proof}

\begin{definition}\label{def:analytic-side}
\uses{def:analytic-k0,def:operator-k-theory}
\notready
For a countable discrete group $\Gamma$, the analytic target of the
coefficient-free reduced Baum--Connes assembly map is
\[
 K_i(C_r^*(\Gamma)),\qquad i=0,1.
\]
The degree-zero group has been constructed above. The degree-one group,
the required functorial interfaces, and the assembly map remain to be constructed.
\end{definition}
