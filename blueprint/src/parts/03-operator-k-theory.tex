\section{Operator K-theory and the analytic side}

\subsection{Projections, stabilization, and $K_0$}
\begin{definition}\label{def:operator-k0}
\notready
For unital $A$, construct $K_0(A)$ as the Grothendieck group of stable
Murray--von Neumann classes of matrix projections, with block-sum addition.
\end{definition}

\subsection{Unitaries, homotopy, and $K_1$}
\begin{definition}\label{def:operator-k1}
\notready
Construct $K_1(A)$ from stable norm-homotopy classes of matrix unitaries,
using $u\mapsto\operatorname{diag}(u,1)$ for stabilization.
\end{definition}

\subsection{Nonunital algebras and functoriality}
\begin{definition}\label{def:operator-k-theory}
\uses{def:operator-k0,def:operator-k1}
\notready
Extend $K_i$ using unitization and the kernel of
$K_i(A^+)\to K_i(\mathbb C)$. Construct the induced maps for
$*$-homomorphisms and establish homotopy invariance. Organize exact
sequences and Bott periodicity as separate theorems, recording which are
needed for the chosen grading convention and the comparison with KK-theory.
\end{definition}

\subsection{The analytic target}
\begin{definition}\label{def:analytic-side}
\uses{def:reduced-group-cstar,def:operator-k-theory}
\notready
The analytic side is $K_i(C_r^*(\Gamma))$, for $i=0,1$.
\end{definition}
